class RelevanceDecision(
    BaseModel
):
    verdict: Literal[
        "relevant",
        "not_relevant",
    ]

    reason: str


# ============================================================
# SYSTEM PROMPT
# ============================================================

SYSTEM_PROMPT = """
Tu es un composant de validation documentaire d'un système RAG.

Ta seule tâche est de déterminer si le PASSAGE DOCUMENTAIRE
fournit réellement une information utile pour comprendre,
interpréter ou contextualiser le FINDING ANALYTIQUE.

Règles strictes :

1. Réponds "relevant" uniquement si le passage fournit
   directement une information applicable au finding.

2. Le simple fait que le passage et le finding parlent
   du même domaine ne suffit PAS.

3. La présence de mots similaires ne suffit PAS.

4. Si une information essentielle du finding n'est pas
   couverte par le passage, choisis "not_relevant".

5. Si tu hésites, choisis "not_relevant".

6. N'invente aucune information absente du passage.

7. Ne déduis pas qu'un passage est pertinent uniquement
   parce qu'il pourrait être intéressant dans une analyse
   plus large.

Le système privilégie l'abstention :
un faux positif documentaire est plus dangereux
qu'un passage pertinent occasionnellement rejeté.

Retourne uniquement la structure JSON demandée.
""".strip()


# ============================================================
# PROMPT
# ============================================================

def build_user_prompt(
    *,
    finding: str,
    passage: str,
) -> str:
    normalized_finding = (
        finding
        .strip()
    )


    normalized_passage = (
        passage
        .strip()
    )


    if not normalized_finding:
        raise ValueError(
            (
                "Le finding ne peut "
                "pas être vide."
            )
        )


    if not normalized_passage:
        raise ValueError(
            (
                "Le passage documentaire "
                "ne peut pas être vide."
            )
        )


    return (
        "FINDING ANALYTIQUE:\n"
        f"{normalized_finding}\n\n"
        "PASSAGE DOCUMENTAIRE:\n"
        f"{normalized_passage}\n\n"
        "Le passage fournit-il directement "
        "un contexte utile au finding ?"
    )


# ============================================================
# CLASSIFICATION
# ============================================================

def classify_relevance(
    *,
    finding: str,
    passage: str,
    model: str = (
        DEFAULT_RELEVANCE_MODEL
    ),
) -> RelevanceDecision:
    user_prompt = (
        build_user_prompt(
            finding=
                finding,

            passage=
                passage,
        )
    )


    try:
        response = client.chat(
            model=
                model,

            messages=[
                {
                    "role":
                        "system",

                    "content":
                        SYSTEM_PROMPT,
                },

                {
                    "role":
                        "user",

                    "content":
                        user_prompt,
                },
            ],

            format=
                RelevanceDecision
                .model_json_schema(),

            options={
                "temperature":
                    0,
            },
        )


    except Exception as error:
        raise RuntimeError(
            (
                "La validation de pertinence "
                "par Ollama a échoué."
            )
        ) from error


    content = (
        response
        .message
        .content
    )


    try:
        return (
            RelevanceDecision
            .model_validate_json(
                content
            )
        )


    except Exception as error:
        raise RuntimeError(
            (
                "Gemma a retourné une réponse "
                "qui ne respecte pas le schéma "
                "de pertinence attendu."
            )
        ) from error

# ============================================================

#
# Important:
# The model no longer decides whether to abstain.
#

        | None
    ) = None

    )

    model: str

# ============================================================
# TEXT NORMALIZATION
# ============================================================

def normalize_whitespace(
    value: str,
) -> str:
    return " ".join(
        value
        .strip()
        .split()
    )


def contains_digit(
    value: str,
) -> bool:
    return bool(
        re.search(
            r"\d",
            value,
        )
    )


# ============================================================
# PROMPT CONSTRUCTION
# ============================================================

def build_user_prompt(
    *,
    finding_text: str,
    accepted_hits: list[
        RagSearchHit
    ],
) -> str:
    finding_text = (
        finding_text
        .strip()
    )


    if not finding_text:
        raise ValueError(
            "Le finding ne peut pas être vide."
        )


    if not accepted_hits:
        raise ValueError(
            (
                "Au moins un passage documentaire "
                "accepté doit être fourni."
            )
        )


    source_blocks: list[
        str
    ] = []


    for hit in accepted_hits:
        source_blocks.append(
            (
                "SOURCE_CHUNK_ID: "
                f"{hit.chunk_id}\n"
                "PASSAGE:\n"
                f"{hit.text}"
            )
        )


    sources_text = (
        "\n\n"
        "========================================\n\n"
        .join(
            source_blocks
        )
    )


    return (
        "FINDING ANALYTIQUE:\n"
        f"{finding_text}\n\n"
        "PASSAGES DOCUMENTAIRES DÉJÀ VALIDÉS "
        "COMME PERTINENTS:\n\n"
        f"{sources_text}\n\n"
        "Extrais maintenant uniquement les claims "
        "documentaires directement soutenus par "
        "ces passages."
    )

# RAW GENERATION

def generate_raw_explanation(

                        "system",

                    "role":

                    "content":

            options={
                "temperature":

    except Exception as error:
        raise RuntimeError(
            (
                "La génération des claims "
                "documentaires a échoué."
            )

    content = (
        response
        .message
        .content

    try:
        return (
            RawRagExplanation

            )

    except Exception as error:

                "Gemma a retourné une réponse "

            )

# ============================================================
# EVIDENCE VALIDATION
# ============================================================

def validate_evidence_quote(
    *,
    evidence_quote: str,
    source_text: str,
) -> str:
    normalized_quote = (
        normalize_whitespace(
            evidence_quote
        )
    )


    normalized_source = (
        normalize_whitespace(
            source_text
        )
    )


    if not normalized_quote:
        raise RuntimeError(
            (
                "Un claim généré contient "
                "une preuve documentaire vide."
            )
        )


    if (
        len(
            normalized_quote
        )
        >
        MAX_EVIDENCE_QUOTE_LENGTH
    ):
        raise RuntimeError(
            (
                "Une preuve documentaire générée "
                "est trop longue."
            )
        )


    if (
        normalized_quote
        not in
        normalized_source
    ):
        raise RuntimeError(
            (
                "Gemma a fourni une preuve "
                "qui n'existe pas textuellement "
                "dans le passage source."
            )
        )


    return normalized_quote


# ============================================================
# CLAIM VALIDATION
# ============================================================

def validate_claim_statement(
    statement: str,
) -> str:
    statement = (
        statement
        .strip()
    )


    if not statement:
        raise RuntimeError(
            "Un claim généré est vide."
        )


    if contains_digit(
        statement
    ):
        raise RuntimeError(
            (
                "Un claim documentaire contient "
                "une valeur numérique interdite."
            )
        )


    if (
        "source_chunk_id"
        in
        statement.lower()
    ):
        raise RuntimeError(
            (
                "Un claim documentaire contient "
                "un identifiant technique de source."
            )
        )


    return statement


# ============================================================
# VERIFIED EXPLANATION CONSTRUCTION
# ============================================================

def build_verified_explanation(
    *,
    claims: list[
        VerifiedGroundedClaim
    ],
) -> str:
    """
    Python constructs the visible explanation from the
    individually verified claims.

    The LLM does not get a second free-form field in which
    unsupported information could be introduced.
    """

    if not claims:
        raise RuntimeError(
            (
                "Aucun claim documentaire "
                "vérifié n'est disponible."
            )
        )


    return " ".join(
        claim.statement

        for claim
        in claims
    )


# ============================================================
# OUTPUT VALIDATION
# ============================================================

def validate_and_build_explanation(
    *,
    raw:
        RawRagExplanation,

    accepted_hits: list[
        RagSearchHit
    ],

    model: str,
) -> VerifiedRagExplanation:
    if not raw.claims:
        raise RuntimeError(
            (
                "Gemma n'a produit "
                "aucun claim documentaire."
            )
        )


    hit_lookup = {
        hit.chunk_id:
            hit

        for hit
        in accepted_hits
    }


    verified_claims: list[
        VerifiedGroundedClaim
    ] = []


    for claim in raw.claims:
        statement = (
            validate_claim_statement(
                claim.statement
            )
        )


        source_chunk_id = (
            claim.source_chunk_id
            .strip()
        )


        source_hit = (
            hit_lookup.get(
                source_chunk_id
            )
        )


        if source_hit is None:
            raise RuntimeError(
                (
                    "Gemma a référencé un chunk "
                    "qui ne fait pas partie des "
                    "accepted_hits."
                )
            )


        evidence_quote = (
            validate_evidence_quote(
                evidence_quote=
                    claim.evidence_quote,

                source_text=
                    source_hit.text,
            )
        )


        citation = (
            RagCitation(
                chunk_id=
                    source_hit.chunk_id,

                filename=
                    source_hit.filename,

                source_locator=
                    source_hit.source_locator,

                page_number=
                    source_hit.page_number,
            )
        )


        verified_claims.append(
            VerifiedGroundedClaim(
                statement=
                    statement,

                evidence_quote=
                    evidence_quote,

                citation=
                    citation,
            )
        )


    explanation = (
        build_verified_explanation(
            claims=
                verified_claims
        )
    )


    if contains_digit(
        explanation
    ):
        raise RuntimeError(
            (
                "L'explication documentaire contient "
                "une valeur numérique interdite."
            )
        )


    return (
        VerifiedRagExplanation(
            status=
                "ready",

            explanation=
                explanation,

            claims=
                verified_claims,

            abstention_reason=
                None,

            model=
                model,
        )
    )


# ============================================================
# PUBLIC FUNCTION
# ============================================================

def generate_grounded_explanation(
    *,
    finding_text: str,
    accepted_hits: list[
        RagSearchHit
    ],
    model: str = (
        DEFAULT_EXPLANATION_MODEL
    ),
) -> VerifiedRagExplanation:
    """
    Generate a documentary explanation only from accepted
    RAG evidence.

    Abstention is deterministic:
    no model call is executed when accepted_hits is empty.

    Once at least one source has passed the relevance gate,
    the explanation model is asked only to extract grounded
    claims. It does not make another relevance decision.
    """

    finding_text = (
        finding_text
        .strip()
    )


    if not finding_text:
        raise ValueError(
            "Le finding ne peut pas être vide."
        )


    # ========================================================
    # DETERMINISTIC ABSTENTION
    # ========================================================

    if not accepted_hits:
        return (
            VerifiedRagExplanation(
                status=
                    "abstained",

                explanation=
                    "",

                claims=[],

                abstention_reason=
                    (
                        "Aucune source documentaire "
                        "validée n'est disponible pour "
                        "cette analyse."
                    ),

                model=
                    model,
            )
        )


    # ========================================================
    # GENERATION
    # ========================================================

    raw = (
        generate_raw_explanation(
            finding_text=
                finding_text,

            accepted_hits=
                accepted_hits,

            model=
                model,
        )
    )


    # ========================================================
    # PYTHON VALIDATION
    # ========================================================

    return (
        validate_and_build_explanation(
            raw=
                raw,

            accepted_hits=
                accepted_hits,

            model=
                model,
        )
    )

# ============================================================
# NORMALIZATION
# ============================================================

def normalize_optional_text(
    value: (
        str
        | None
    ),
) -> (
    str
    | None
):
    if value is None:
        return None


    normalized = (
        value
        .strip()
    )


    if not normalized:
        return None


    return normalized


# ============================================================
# SUMMARY EXTRACTION
# ============================================================

def extract_summary_items(
    finding: Any,
) -> list[
    str
]:
    summary = getattr(
        finding,
        "summary",
        None,
    )


    if summary is None:
        return []


    if isinstance(
        summary,
        str,
    ):
        normalized = (
            summary
            .strip()
        )


        return (
            [
                normalized
            ]
            if normalized
            else []
        )


    if isinstance(
        summary,
        (
            list,
            tuple,
        ),
    ):
        return [
            str(
                item
            ).strip()

            for item
            in summary[
                :2
            ]

            if str(
                item
            ).strip()
        ]


    return []


# ============================================================
# FINDING RETRIEVAL QUERY
# ============================================================

def build_finding_retrieval_query(
    *,
    finding: Any,
) -> str:
    """
    Build a semantic retrieval query from the finding itself.

    The global business objective is deliberately excluded
    from vector retrieval because diagnostics showed that
    it could dominate the finding-specific semantic signal.
    """

    parts: list[
        str
    ] = []


    title = str(
        getattr(
            finding,
            "title",
            "",
        )
        or ""
    ).strip()


    if title:
        parts.append(
            title
        )


    family = str(
        getattr(
            finding,
            "family",
            "",
        )
        or ""
    ).strip()


    if family:
        parts.append(
            (
                "Type d'analyse : "
                f"{family}"
            )
        )


    datasets = (
        getattr(
            finding,
            "datasets",
            [],
        )
        or []
    )


    dataset_names = [
        str(
            dataset
        ).strip()

        for dataset
        in datasets

        if str(
            dataset
        ).strip()
    ]


    if dataset_names:
        parts.append(
            (
                "Données concernées : "
                +
                ", ".join(
                    dataset_names
                )
            )
        )


    summary_items = (
        extract_summary_items(
            finding
        )
    )


    if summary_items:
        parts.append(
            (
                "Résultats calculés : "
                +
                " ".join(
                    summary_items
                )
            )
        )


    parts.append(
        (
            "Chercher le contexte documentaire "
            "directement pertinent pour comprendre "
            "cette analyse : définition métier, "
            "interprétation, contrainte, règle ou "
            "recommandation liée précisément aux "
            "variables et au phénomène étudiés."
        )
    )


    query = "\n".join(
        parts
    ).strip()


    if not query:
        raise ValueError(
            (
                "Impossible de construire "
                "une requête documentaire "
                "pour cette analyse."
            )
        )


    return query


# ============================================================
# FINDING TEXT FOR RELEVANCE / EXPLANATION
# ============================================================

def build_finding_relevance_text(
    *,
    finding: Any,
) -> str:
    """
    Build the compact analytical statement used by both:

    - the relevance gate;
    - the grounded explanation component.

    It contains deterministic analysis information only.
    """

    parts: list[
        str
    ] = []


    title = str(
        getattr(
            finding,
            "title",
            "",
        )
        or ""
    ).strip()


    if title:
        parts.append(
            title
        )


    summary_items = (
        extract_summary_items(
            finding
        )
    )


    if summary_items:
        parts.append(
            (
                "Résultats calculés : "
                +
                " ".join(
                    summary_items
                )
            )
        )


    finding_text = "\n".join(
        parts
    ).strip()


    if not finding_text:
        raise ValueError(
            (
                "Impossible de construire "
                "le finding utilisé pour "
                "la validation documentaire."
            )
        )


    return finding_text


# ============================================================
# SAFE EXPLANATION FALLBACK
# ============================================================

def build_safe_explanation_fallback(
    *,
    model: str,
    reason: str,
) -> VerifiedRagExplanation:
    """
    Fail closed.

    If grounded generation fails validation, DataLens does not
    fall back to unconstrained text generation.
    """

    return (
        VerifiedRagExplanation(
            status=
                "abstained",

            explanation=
                "",

            claims=[],

            abstention_reason=
                reason,

            model=
                model,
        )
    )


# ============================================================
# CONTEXT RETRIEVAL + RELEVANCE + EXPLANATION
# ============================================================

def retrieve_context_for_report(
    *,
    report:
        UnifiedAnalysisReport,

    ingestion:
        DocumentIngestionReport,

    objective: (
        str
        | None
    ) = None,

    top_k: int = (
        DEFAULT_TOP_K
    ),

    model: str = (
        DEFAULT_EMBEDDING_MODEL
    ),

    relevance_model: str = (
        DEFAULT_RELEVANCE_MODEL
    ),

    explanation_model: str = (
        DEFAULT_EXPLANATION_MODEL
    ),
) -> RagContextReport:
    if not ingestion.chunks:
        raise ValueError(
            (
                "Aucun passage documentaire "
                "n'est disponible."
            )
        )


    if (
        top_k
        <
        1
    ):
        raise ValueError(
            (
                "top_k doit être supérieur "
                "ou égal à 1."
            )
        )


    findings = list(
        report.main_findings
    )


    effective_top_k = min(
        top_k,
        MAX_TOP_K,
        len(
            ingestion.chunks
        ),
    )


    normalized_objective = (
        normalize_optional_text(
            objective
        )
    )


    # ========================================================
    # EMPTY REPORT
    # ========================================================

    if not findings:
        return RagContextReport(
            objective=
                normalized_objective,

            document_count=
                ingestion.document_count,

            chunk_count=
                ingestion.chunk_count,

            finding_count=
                0,

            top_k=
                effective_top_k,

            model=
                model,

            relevance_model=
                relevance_model,

            explanation_model=
                explanation_model,

            validated_candidate_count=
                0,

            accepted_hit_count=
                0,

            accepted_finding_count=
                0,

            abstained_finding_count=
                0,

            explanation_ready_count=
                0,

            explanation_abstained_count=
                0,

            explanation_error_count=
                0,

            contexts=[],
        )


    # ========================================================
    # 1. RETRIEVAL QUERIES
    # ========================================================

    finding_queries = [
        build_finding_retrieval_query(
            finding=
                finding,
        )

        for finding
        in findings
    ]


    # ========================================================
    # 2. FINDING TEXTS
    # ========================================================

    relevance_finding_texts = [
        build_finding_relevance_text(
            finding=
                finding,
        )

        for finding
        in findings
    ]


    # ========================================================
    # 3. EMBEDDINGGEMMA INPUTS
    # ========================================================

    prepared_queries = [
        prepare_query_for_embedding(
            query
        )

        for query
        in finding_queries
    ]


    prepared_documents = [
        prepare_document_for_embedding(
            chunk
        )

        for chunk
        in ingestion.chunks
    ]


    # ========================================================
    # 4. SINGLE EMBEDDING BATCH
    # ========================================================

    embedding_inputs = [
        *prepared_queries,
        *prepared_documents,
    ]


    embeddings = (
        embed_text_batch(
            embedding_inputs,
            model=
                model,
        )
    )


    query_count = len(
        prepared_queries
    )


    query_embeddings = (
        embeddings[
            :query_count
        ]
    )


    document_embeddings = (
        embeddings[
            query_count:
        ]
    )


    if (
        len(
            document_embeddings
        )
        !=
        len(
            ingestion.chunks
        )
    ):
        raise RuntimeError(
            (
                "Le nombre d'embeddings "
                "documentaires ne correspond "
                "pas au nombre de chunks."
            )
        )


    # ========================================================
    # COUNTERS
    # ========================================================

    contexts: list[
        FindingRagContext
    ] = []


    validated_candidate_count = 0

    accepted_hit_count = 0

    accepted_finding_count = 0

    abstained_finding_count = 0


    explanation_ready_count = 0

    explanation_abstained_count = 0

    explanation_error_count = 0


    # ========================================================
    # 5. PROCESS EACH FINDING
    # ========================================================

    for (
        finding_index,
        (
            finding,
            query,
            relevance_finding_text,
            query_embedding,
        ),
    ) in enumerate(
        zip(
            findings,
            finding_queries,
            relevance_finding_texts,
            query_embeddings,
            strict=True,
        ),
        start=1,
    ):
        scored_chunks: list[
            tuple[
                float,
                Any,
            ]
        ] = []


        # ====================================================
        # 5A. VECTOR RETRIEVAL
        # ====================================================

        for (
            chunk,
            document_embedding,
        ) in zip(
            ingestion.chunks,
            document_embeddings,
            strict=True,
        ):
            score = (
                cosine_similarity(
                    query_embedding,
                    document_embedding,
                )
            )


            scored_chunks.append(
                (
                    score,
                    chunk,
                )
            )


        scored_chunks.sort(
            key=lambda item: (
                -item[
                    0
                ],
                item[
                    1
                ].filename,
                item[
                    1
                ].chunk_index,
            )
        )


        selected = (
            scored_chunks[
                :effective_top_k
            ]
        )


        hits = [
            build_search_hit(
                rank=
                    rank,

                score=
                    score,

                chunk=
                    chunk,
            )

            for (
                rank,
                (
                    score,
                    chunk,
                ),
            ) in enumerate(
                selected,
                start=1,
            )
        ]


        # ====================================================
        # 5B. RELEVANCE GATE
        # ====================================================

        relevance_decisions: list[
            RagHitRelevanceDecision
        ] = []


        accepted_hits: list[
            RagSearchHit
        ] = []


        for hit in hits:
            decision = (
                classify_relevance(
                    finding=
                        relevance_finding_text,

                    passage=
                        hit.text,

                    model=
                        relevance_model,
                )
            )


            validated_candidate_count += 1


            relevance_decisions.append(
                RagHitRelevanceDecision(
                    rank=
                        hit.rank,

                    chunk_id=
                        hit.chunk_id,

                    filename=
                        hit.filename,

                    source_locator=
                        hit.source_locator,

                    score=
                        hit.score,

                    verdict=
                        decision.verdict,

                    reason=
                        decision.reason,
                )
            )


            if (
                decision.verdict
                ==
                "relevant"
            ):
                accepted_hits.append(
                    hit
                )


                accepted_hit_count += 1


        # ====================================================
        # 5C. DOCUMENTARY ABSTENTION
        # ====================================================

        abstained = (
            len(
                accepted_hits
            )
            ==
            0
        )


        if abstained:
            abstained_finding_count += 1


            abstention_reason = (
                "Aucun passage documentaire "
                "récupéré n'a été jugé "
                "suffisamment pertinent pour "
                "contextualiser cette analyse."
            )


        else:
            accepted_finding_count += 1


            abstention_reason = None


        # ====================================================
        # 5D. GROUNDED EXPLANATION
        # ====================================================

        explanation_error: (
            str
            | None
        ) = None


        try:
            explanation = (
                generate_grounded_explanation(
                    finding_text=
                        relevance_finding_text,

                    accepted_hits=
                        accepted_hits,

                    model=
                        explanation_model,
                )
            )


        except RuntimeError as error:
            explanation_error_count += 1


            explanation_error = str(
                error
            )


            explanation = (
                build_safe_explanation_fallback(
                    model=
                        explanation_model,

                    reason=
                        (
                            "L'explication documentaire "
                            "n'a pas pu être validée de "
                            "manière suffisamment sûre."
                        ),
                )
            )


        if (
            explanation.status
            ==
            "ready"
        ):
            explanation_ready_count += 1


        else:
            explanation_abstained_count += 1


        # ====================================================
        # 5E. FINDING IDENTITY
        # ====================================================

        raw_analysis_id = getattr(
            finding,
            "analysis_id",
            None,
        )


        analysis_id = (
            str(
                raw_analysis_id
            ).strip()

            if raw_analysis_id
            else (
                "finding:"
                f"{finding_index:04d}"
            )
        )


        title = str(
            getattr(
                finding,
                "title",
                "",
            )
            or analysis_id
        )


        family = str(
            getattr(
                finding,
                "family",
                "",
            )
            or "unknown"
        )


        # ====================================================
        # 5F. CONTEXT RESULT
        # ====================================================

        contexts.append(
            FindingRagContext(
                analysis_id=
                    analysis_id,

                title=
                    title,

                family=
                    family,

                query=
                    query,

                relevance_finding_text=
                    relevance_finding_text,

                hits=
                    hits,

                relevance_decisions=
                    relevance_decisions,

                accepted_hits=
                    accepted_hits,

                abstained=
                    abstained,

                abstention_reason=
                    abstention_reason,

                explanation=
                    explanation,

                explanation_error=
                    explanation_error,
            )
        )


    # ========================================================
    # 6. COMPLETE REPORT
    # ========================================================

    return RagContextReport(
        objective=
            normalized_objective,

        document_count=
            ingestion.document_count,

        chunk_count=
            ingestion.chunk_count,

        finding_count=
            len(
                contexts
            ),

        top_k=
            effective_top_k,

        model=
            model,

        relevance_model=
            relevance_model,

        explanation_model=
            explanation_model,

        validated_candidate_count=
            validated_candidate_count,

        accepted_hit_count=
            accepted_hit_count,

        accepted_finding_count=
            accepted_finding_count,

        abstained_finding_count=
            abstained_finding_count,

        explanation_ready_count=
            explanation_ready_count,

        explanation_abstained_count=
            explanation_abstained_count,

        explanation_error_count=
            explanation_error_count,

        contexts=
            contexts,
    )
